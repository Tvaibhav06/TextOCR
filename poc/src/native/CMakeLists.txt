# CMakeLists.txt — OpenCVProcessor native module (Android)
#
# Track 3: OpenCV is now ENABLED by default.
#
# ─── Integration into android/app/build.gradle ───────────────────────────────
#
#   Download OpenCV Android SDK from https://opencv.org/releases/
#   (tested with 4.8.0 and 4.9.0).
#   Extract to e.g. D:/libs/OpenCV-android-sdk/
#
#   In android/app/build.gradle, add inside android {}:
#
#   externalNativeBuild {
#     cmake {
#       path "../../src/native/CMakeLists.txt"
#       version "3.22.1"
#     }
#   }
#   defaultConfig {
#     externalNativeBuild {
#       cmake {
#         arguments "-DANDROID_STL=c++_shared",
#                   "-DREACT_NATIVE_DIR=${rootDir}/../node_modules/react-native",
#                   "-DOPENCV_DIR=D:/libs/OpenCV-android-sdk/sdk/native/jni"
#         abiFilters "arm64-v8a", "x86_64"
#       }
#     }
#   }
#
#   Also add to android/app/src/main/jniLibs/<abi>/ the OpenCV shared libs:
#     libopencv_java4.so  (or the prebuilt .so from the SDK)
#
# ─── Disable OpenCV (Track 2 compatibility / CI without SDK) ─────────────────
#   cmake -DENABLE_OPENCV=OFF ...

cmake_minimum_required(VERSION 3.22.1)
project(RNOpenCVProcessor)

# ─── Build options ─────────────────────────────────────────────────────────────
option(ENABLE_OPENCV "Link OpenCV Android SDK for real image processing" ON)
set(REACT_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../node_modules/react-native"
    CACHE PATH "Path to react-native node_modules")

# OPENCV_DIR must point to the sdk/native/jni directory of the OpenCV Android SDK.
# Example: /path/to/OpenCV-android-sdk/sdk/native/jni
set(OPENCV_DIR "" CACHE PATH "Path to OpenCV Android SDK sdk/native/jni directory")

# ─── Source files ──────────────────────────────────────────────────────────────
# Note: We use standard JNI — no JSI or TurboModule headers needed.
add_library(
    RNOpenCVProcessor
    SHARED
    "${CMAKE_CURRENT_SOURCE_DIR}/OpenCVProcessor.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/OnLoad.cpp"
)

target_include_directories(RNOpenCVProcessor PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}"
)

# ─── Standard Android system libs ─────────────────────────────────────────────
find_library(log-lib log)

target_link_libraries(RNOpenCVProcessor
    ${log-lib}
)

# ─── C++ standard ─────────────────────────────────────────────────────────────
target_compile_features(RNOpenCVProcessor PRIVATE cxx_std_17)

# ─── OpenCV ───────────────────────────────────────────────────────────────────
if(ENABLE_OPENCV)
    if("${OPENCV_DIR}" STREQUAL "")
        message(FATAL_ERROR
            "ENABLE_OPENCV=ON but OPENCV_DIR is not set.\n"
            "Download OpenCV Android SDK from https://opencv.org/releases/\n"
            "and pass -DOPENCV_DIR=/path/to/OpenCV-android-sdk/sdk/native/jni\n"
            "Or disable with -DENABLE_OPENCV=OFF for stub mode.")
    endif()

    set(OpenCV_DIR "${OPENCV_DIR}")
    find_package(OpenCV REQUIRED core imgproc photo imgcodecs)

    target_include_directories(RNOpenCVProcessor PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(RNOpenCVProcessor ${OpenCV_LIBS})
    target_compile_definitions(RNOpenCVProcessor PRIVATE OPENCV_ENABLED=1)

    message(STATUS "OpenCV ${OpenCV_VERSION} linked from ${OPENCV_DIR}")
else()
    # Stub mode: remove OpenCV includes from the header to avoid missing symbols.
    # In this mode OpenCVProcessor.cpp will not compile unless you guard includes.
    message(WARNING
        "OpenCV DISABLED — native module will not compile with real cv:: calls.\n"
        "This mode is only for JS-only test builds. Set -DENABLE_OPENCV=ON for real builds.")
    target_compile_definitions(RNOpenCVProcessor PRIVATE OPENCV_ENABLED=0)
endif()
